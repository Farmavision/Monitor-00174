async function controlloAperturaCalendario(calendarId) { if (!calendarId) { impostaErroreCalendario("ID Calendario non configurato nel DB"); return; } const adesso = new Date(); const timeMin = adesso.toISOString(); // Scarichiamo fino a 10 eventi per poter scavalcare ponti o ferie lunghe const url = `https://www.googleapis.com/calendar/v3/calendars/${encodeURIComponent(calendarId)}/events?timeMin=${timeMin}&maxResults=10&singleEvents=true&orderBy=startTime&key=${GOOGLE_API_KEY}`; try { const res = await fetch(url); if (!res.ok) throw new Error("Risposta non valida dalle API Calendar"); const data = await res.json(); const headerEl = document.getElementById('header'); const statoEl = document.getElementById('stato-testo'); const dettEl = document.getElementById('orario-dettaglio'); if (data.items && data.items.length > 0) { // 1. CERCHIAMO L'EVENTO IN CORSO IN QUESTO PRECISO SECONDO const eventoInCorso = data.items.find(ev => { const start = new Date(ev.start.dateTime || ev.start.date); const end = new Date(ev.end.dateTime || ev.end.date); return start <= adesso && end > adesso; }); // 2. CERCHIAMO IL PRIMO EVENTO "APERTO" NEL FUTURO (LA VERA PROSSIMA RIAPERTURA) const prossimaApertura = data.items.find(ev => { const titolo = ev.summary ? ev.summary.toUpperCase() : ""; const start = new Date(ev.start.dateTime || ev.start.date); return start > adesso && titolo.includes("APERTO"); }); // FUNZIONE INTERNA PER FORMATTARE LA DATA DI RIAPERTURA IN ITALIANO CORRETTO const calcolaTestoRiapertura = (aperturaEv) => { if (!aperturaEv) return "CONSULTARE GLI ORARI DI TURNO QUI SOTTO"; const start = new Date(aperturaEv.start.dateTime || aperturaEv.start.date); const ora = start.toLocaleTimeString('it-IT', { hour: '2-digit', minute: '2-digit' }); // Se riapre oggi if (start.toDateString() === adesso.toDateString()) { return "RIAPRE OGGI ALLE ORE " + ora; } // Se riapre domani const domani = new Date(adesso); domani.setDate(domani.getDate() + 1); if (start.toDateString() === domani.toDateString()) { return "RIAPRE DOMANI ALLE ORE " + ora; } // Se riapre in un giorno successivo della settimana const giornoSettimana = start.toLocaleDateString('it-IT', { weekday: 'long' }).toUpperCase(); const dataMese = start.toLocaleDateString('it-IT', { day: '2-digit', month: '2-digit' }); return `RIAPRE ${giornoSettimana} ${dataMese} ALLE ORE ${ora}`; }; // ========================================================= // GESTIONE DEI CASI DI VISUALIZZAZIONE // ========================================================= if (eventoInCorso) { const titoloInCorso = eventoInCorso.summary ? eventoInCorso.summary.toUpperCase() : ""; if (titoloInCorso.includes("APERTO")) { // CASO A: Siamo aperti adesso (Orario regolare) headerEl.className = "aperto"; // VERDE statoEl.innerText = "APERTO"; statoEl.className = "testo-aperto"; const endEv = new Date(eventoInCorso.end.dateTime || eventoInCorso.end.date); const oraFineStr = endEv.toLocaleTimeString('it-IT', { hour: '2-digit', minute: '2-digit' }); dettEl.innerText = "APERTO FINO ALLE ORE " + oraFineStr; } else { // CASO B: C'è un evento ora, ma è di chiusura (es. siamo in "FERIE" o "CHIUSO PER ASSEMBLEA") headerEl.className = "chiuso"; // ROSSO statoEl.innerText = titoloInCorso; // Mostra la causale esatta scritta su Google Calendar statoEl.className = "testo-chiuso"; // Cerca il primo "APERTO" disponibile e calcola quando aprirà dettEl.innerText = calcolaTestoRiapertura(prossimaApertura); } } else { // CASO C: Nessun evento in corso (La farmacia è chiusa, es. di notte o pausa pranzo) headerEl.className = "chiuso"; // ROSSO statoEl.className = "testo-chiuso"; // Controlliamo se c'è un evento che inizierà a breve (il primo cronologicamente nel futuro) const primoFuturoGenerico = data.items.find(ev => new Date(ev.start.dateTime || ev.start.date) > adesso); const titoloProssimo = primoFuturoGenerico && primoFuturoGenerico.summary ? primoFuturoGenerico.summary.toUpperCase() : ""; if (titoloProssimo && !titoloProssimo.includes("APERTO")) { // Se il prossimo evento imminente è una chiusura (es. stasera è chiuso e il calendario dice che domani iniziano le "FERIE") // usiamo il titolo dell'evento come avviso anticipato. statoEl.innerText = titoloProssimo; } else { // Chiusura ordinaria notturna/serale statoEl.innerText = "ATTUALMENTE CHIUSA"; } // Indica tassativamente quando ci sarà il primo evento "APERTO" dettEl.innerText = calcolaTestoRiapertura(prossimaApertura); } } else { // Calendario totalmente vuoto (Nessun evento trovato) headerEl.className = "chiuso"; statoEl.innerText = "ATTUALMENTE CHIUSA"; statoEl.className = "testo-chiuso"; dettEl.innerText = "CONSULTARE GLI ORARI DI TURNO QUI SOTTO"; } } catch (error) { console.error("Errore Calendario:", error); impostaErroreCalendario("Errore Sincronizzazione Live Orari"); } }
